    @GET("Luchitxo/listas-iptv/refs/heads/main/tv_vod.json")

    companion object {
        private const val BASE_URL = "https://raw.githubusercontent.com/"

                .baseUrl(BASE_URL)
                .client(okHttpClient)
                .addConverterFactory(GsonConverterFactory.create())
                .build()
                .create(ApiService::class.java)
        }
    }

package com.example.miiptvvodtv.data

import com.example.miiptvvodtv.Channel
import java.util.UUID

object M3UParser {

    fun parse(m3uContent: String): List<Channel> {
        val channels = mutableListOf<Channel>()
        val lines = m3uContent.lines()

        var currentName = "Canal sin nombre"
        var currentCategory = "General"
        var currentLogo = ""
        var currentId = ""
        var currentHeaders = mutableMapOf<String, String>()
        var pendingExtinf = false

        val extinfRegex = Regex("""#EXTINF:(?:-?\d+)?\s*(.*)""", RegexOption.IGNORE_CASE)
        val attrRegex = Regex("""([a-zA-Z0-9_-]+)=["']?([^"']*)["']?""")

        for (line in lines) {
            val trimmed = line.trim()
            if (trimmed.isEmpty()) continue

            when {
                trimmed.startsWith("#EXTM3U", ignoreCase = true) -> {
                    // Header M3U
                }
                trimmed.startsWith("#EXTINF", ignoreCase = true) -> {
                    pendingExtinf = true
                    currentName = "Canal sin nombre"
                    currentCategory = "General"
                    currentLogo = ""
                    currentId = ""
                    currentHeaders.clear()

                    val match = extinfRegex.find(trimmed)
                    if (match != null) {
                        val paramsAndTitle = match.groupValues[1]

                        attrRegex.findAll(paramsAndTitle).forEach { attrMatch ->
                            val key = attrMatch.groupValues[1].lowercase()
                            val value = attrMatch.groupValues[2]
                            when {
                                key.contains("tvg-id") -> currentId = value
                                key.contains("tvg-name") && currentName == "Canal sin nombre" -> currentName = value
                                key.contains("tvg-logo") -> currentLogo = value
                                key.contains("group-title") -> currentCategory = value
                            }
                        }

                        val commaIndex = paramsAndTitle.lastIndexOf(',')
                        if (commaIndex != -1 && commaIndex < paramsAndTitle.length - 1) {
                            val title = paramsAndTitle.substring(commaIndex + 1).trim()
                            if (title.isNotEmpty()) {
                                currentName = title
                            }
                        }
                    }
                }
                trimmed.startsWith("#EXTVLCOPT", ignoreCase = true) -> {
                    val optContent = trimmed.substringAfter("#EXTVLCOPT:", "").trim()
                    if (optContent.contains("http-user-agent=", ignoreCase = true)) {
                        val ua = optContent.substringAfter("http-user-agent=", "").trim()
                        if (ua.isNotEmpty()) {
                            currentHeaders["User-Agent"] = ua
                        }
                    } else if (optContent.contains("http-referer=", ignoreCase = true) || optContent.contains("http-referrer=", ignoreCase = true)) {
                        val ref = optContent.substringAfter("http-referer=", "").let { if (it == optContent) optContent.substringAfter("http-referrer=", "") else it }.trim()
                        if (ref.isNotEmpty()) {
                            currentHeaders["Referer"] = ref
                        }
                    }
                }
                !trimmed.startsWith("#") && pendingExtinf -> {
                    val url = trimmed
                    val finalId = if (currentId.isNotBlank()) currentId else UUID.nameUUIDFromBytes(url.toByteArray()).toString()

                    channels.add(
                        Channel(
                            id = finalId,
                            name = currentName,
                            category = currentCategory.ifEmpty { "General" },
                            logo = currentLogo,
                            url = url,
                            headers = if (currentHeaders.isNotEmpty()) currentHeaders.toMap() else null
                        )
                    )
                    pendingExtinf = false
                }
            }
        }

        return channels
    }
}

package com.example.miiptvvodtv.data

import android.content.Context
import android.util.Log
import coil.imageLoader
import coil.request.CachePolicy
import coil.request.ImageRequest
import com.example.miiptvvodtv.ApiService
import com.example.miiptvvodtv.Channel
import com.example.miiptvvodtv.MediaResponse
import com.google.gson.Gson
import kotlinx.coroutines.CancellationException
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import java.io.File
import java.util.concurrent.TimeUnit
import okhttp3.OkHttpClient
import okhttp3.Request

data class FetchResult(
    val data: MediaResponse,
    val isFromCache: Boolean
)

data class CachedMediaWrapper(
    val timestamp: Long,
    val data: MediaResponse
)

class MediaRepository(private val context: Context) {

    private val gson = Gson()
    private val cacheFile: File by lazy {
        File(context.cacheDir, "media_cache_v2.json")
    }

    fun getCachedContent(): MediaResponse? {
        return try {
            if (cacheFile.exists()) {
                val jsonString = cacheFile.readText()
                val wrapper = gson.fromJson(jsonString, CachedMediaWrapper::class.java)
                wrapper?.data
            } else {
                val prefs = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE)
                val oldJson = prefs.getString(KEY_CACHE_JSON, null)
                if (!oldJson.isNullOrEmpty()) {
                    gson.fromJson(oldJson, MediaResponse::class.java)
                } else {
                    null
                }
            }
        } catch (e: Exception) {
            if (e is CancellationException) throw e
            Log.e(TAG, "Error al leer datos de la caché local", e)
            null
        }
    }

    suspend fun fetchAndSaveRemoteContent(
        onProgress: (percent: Int, status: String) -> Unit = { _, _ -> }
    ): MediaResponse = withContext(Dispatchers.IO) {
        onProgress(10, "Actualizando desde la nube...")

        val rawText = try {
            val client = OkHttpClient.Builder()
                .connectTimeout(20, TimeUnit.SECONDS)
                .readTimeout(20, TimeUnit.SECONDS)
                .build()
            val request = Request.Builder()
                .url("https://raw.githubusercontent.com/Luchitxo/listas-iptv/refs/heads/main/tv_vod.json")
                .build()
            client.newCall(request).execute().use { response ->
                response.body?.string() ?: ""
            }
        } catch (e: Exception) {
            if (e is CancellationException) throw e
            Log.e(TAG, "Error descargando contenido crudo, intentando fallback Retrofit", e)
            val retrofitRes = ApiService.instance.getContent()
            return@withContext processMediaResponse(retrofitRes, onProgress)
        }

        val channels = if (rawText.trim().startsWith("#EXTM3U", ignoreCase = true) || rawText.contains("#EXTINF:")) {
            M3UParser.parse(rawText)
        } else {
            val resp = gson.fromJson(rawText, MediaResponse::class.java)
            resp?.channels ?: emptyList()
        }

        processChannels(channels, onProgress)
    }

    private fun processMediaResponse(
        response: MediaResponse,
        onProgress: (percent: Int, status: String) -> Unit
    ): MediaResponse {
        return processChannels(response.channels ?: emptyList(), onProgress)
    }

    private fun processChannels(
        rawChannels: List<Channel>,
        onProgress: (percent: Int, status: String) -> Unit
    ): MediaResponse {
        if (rawChannels.isEmpty()) {
            throw IllegalStateException("El servidor devolvió una lista vacía de canales.")
        }

        onProgress(50, "Validando y filtrando canales...")
        val validChannels = rawChannels.filter {
            !it.id.isBlank() && !it.url.isBlank()
        }.map { channel ->
            channel.copy(
                name = if (channel.name.isBlank()) "Canal sin nombre" else channel.name,
                category = if (channel.category.isBlank()) "General" else channel.category
            )
        }.distinctBy { it.id }

        if (validChannels.isEmpty()) {
            throw IllegalStateException("No hay canales válidos tras la validación.")
        }

        val sanitizedResponse = MediaResponse(channels = validChannels)

        onProgress(80, "Guardando caché en archivo...")
        val wrapper = CachedMediaWrapper(
            timestamp = System.currentTimeMillis(),
            data = sanitizedResponse
        )
        cacheFile.writeText(gson.toJson(wrapper))

        preloadImages(validChannels, getFavorites())

        onProgress(100, "¡Actualizado con éxito!")
        return sanitizedResponse
    }

    suspend fun getMediaContent(
        onProgress: (percent: Int, status: String) -> Unit = { _, _ -> }
    ): FetchResult = withContext(Dispatchers.IO) {
        val cached = getCachedContent()
        if (cached != null && !cached.channels.isNullOrEmpty()) {
            FetchResult(data = cached, isFromCache = true)
        } else {
            val freshData = fetchAndSaveRemoteContent(onProgress)
            FetchResult(data = freshData, isFromCache = false)
        }
    }

    private fun preloadImages(channels: List<Channel>, favoriteIds: Set<String>) {
        val imageLoader = context.imageLoader
        val favoriteChannels = channels.filter { favoriteIds.contains(it.id) }
        val otherChannels = channels.filter { !favoriteIds.contains(it.id) }
        val channelsToPreload = (favoriteChannels + otherChannels).distinctBy { it.id }.take(50)

        channelsToPreload.forEach { channel ->
            if (channel.logo.isNotEmpty()) {
                val request = ImageRequest.Builder(context)
                    .data(channel.logo)
                    .size(160, 160)
                    .memoryCachePolicy(CachePolicy.ENABLED)
                    .diskCachePolicy(CachePolicy.ENABLED)
                    .build()
                imageLoader.enqueue(request)
            }
        }
    }

    fun getFavorites(): Set<String> {
        val prefs = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE)
        return prefs.getStringSet(KEY_FAVORITES, emptySet()) ?: emptySet()
    }

    fun toggleFavorite(id: String): Set<String> {
        val prefs = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE)
        val currentFavorites = getFavorites().toMutableSet()
        if (currentFavorites.contains(id)) {
            currentFavorites.remove(id)
        } else {
            currentFavorites.add(id)
        }
        prefs.edit().putStringSet(KEY_FAVORITES, currentFavorites).apply()
        return currentFavorites
    }

    fun getPreferredUrl(channelId: String): String? {
        val prefs = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE)
        return prefs.getString("preferred_url_$channelId", null)
    }

    fun savePreferredUrl(channelId: String, url: String) {
        val prefs = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE)
        prefs.edit().putString("preferred_url_$channelId", url).apply()
    }

    fun clearPreferredUrl(channelId: String) {
        val prefs = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE)
        prefs.edit().remove("preferred_url_$channelId").apply()
    }

    companion object {
        private const val TAG = "MediaRepository"
        private const val PREFS_NAME = "mi_iptv_vod_prefs"
        private const val KEY_CACHE_JSON = "cached_media_json"
        private const val KEY_FAVORITES = "favorite_ids"
    }
}

package com.example.miiptvvodtv.utils

import android.app.UiModeManager
import android.content.Context
import android.content.pm.PackageManager
import android.content.res.Configuration

object DeviceUtils {
    fun isTvDevice(context: Context): Boolean {
        val uiModeManager = context.getSystemService(Context.UI_MODE_SERVICE) as? UiModeManager
        if (uiModeManager?.currentModeType == Configuration.UI_MODE_TYPE_TELEVISION) {
            return true
        }
        return context.packageManager.hasSystemFeature(PackageManager.FEATURE_LEANBACK)
    }
}

private const val TAG = "VideoPlayer"

private enum class TrackStatus {
    OK, NO_VIDEO, NO_AUDIO, AUDIO_UNSUPPORTED
}

private fun checkTracks(player: ExoPlayer): TrackStatus {
    val tracks = player.currentTracks
    var hasSupportedVideo = false
    var hasSupportedAudio = false
    val audioMimeTypes = mutableListOf<String>()

    for (group in tracks.groups) {
        if (group.type == C.TRACK_TYPE_VIDEO) {
            for (i in 0 until group.length) {
                if (group.isTrackSupported(i)) {
                    hasSupportedVideo = true
                    break
                }
            }
        } else if (group.type == C.TRACK_TYPE_AUDIO) {
            for (i in 0 until group.length) {
                if (group.isTrackSupported(i)) {
                    hasSupportedAudio = true
                    val format = group.getTrackFormat(i)
                    format.sampleMimeType?.let { audioMimeTypes.add(it) }
                }
            }
        }
    }

    if (!hasSupportedVideo) {
        return TrackStatus.NO_VIDEO
    }
    if (!hasSupportedAudio) {
        return TrackStatus.NO_AUDIO
    }

    if (audioMimeTypes.isNotEmpty() && audioMimeTypes.all { it == MimeTypes.AUDIO_AC3 || it == MimeTypes.AUDIO_E_AC3 }) {
        return TrackStatus.AUDIO_UNSUPPORTED
    }

    return TrackStatus.OK
}

fun VideoPlayer(
    context: Context,
    videoUrl: String,
    channelName: String = "Canal en Vivo",

    var isInactivePrompt by remember { mutableStateOf(false) }
    var isBuffering by remember { mutableStateOf(true) }

    var alternativeIndex by remember { mutableIntStateOf(-1) }
    var currentActiveUrl by remember { mutableStateOf(videoUrl) }
    val maxAutoRetries = 4

    var isPlaying by remember { mutableStateOf(true) }
    var showControlsOverlay by remember { mutableStateOf(true) }

    val playerFocusRequester = remember { FocusRequester() }
    val retryFocusRequester = remember { FocusRequester() }
    val inactiveFocusRequester = remember { FocusRequester() }
    val coroutineScope = rememberCoroutineScope()
    val lifecycleOwner = LocalLifecycleOwner.current

    val currentChannelIndex = remember(currentChannelId, channelsList) {
        channelsList.indexOfFirst { it.id == currentChannelId }
    }

    BackHandler {
        onBack()
    }

    LaunchedEffect(showControlsOverlay) {

            delay(50)
            playerFocusRequester.requestFocus()
        }

        val requestHeaders = mutableMapOf(

        val effectiveUserAgent = requestHeaders["User-Agent"] ?: defaultUserAgent

        val httpDataSourceFactory = DefaultHttpDataSource.Factory()
            .setUserAgent(effectiveUserAgent)

            .setConnectTimeoutMs(20000)
            .setReadTimeoutMs(20000)

            .apply {

                            .setMimeType(MimeTypes.APPLICATION_MPD)
                            .build()

                }

                    Log.i("TrackCheck", "Estado $status detectado. Disparando fallback a alternativa...")
                    switchToNextAlternative()
                }
                TrackStatus.NO_AUDIO -> {
                    Log.i("TrackCheck", "Canal sin audio detectado (NO_AUDIO). Se mantiene reproduciendo.")
                    if (exoPlayer.playbackState == Player.STATE_READY && exoPlayer.isPlaying) {
                        onPlaybackStable(currentActiveUrl)
                    }
                }
                TrackStatus.OK -> {
                    Log.i("TrackCheck", "Pistas correctas (OK).")
                    if (exoPlayer.playbackState == Player.STATE_READY && exoPlayer.isPlaying) {
                        Log.i("TrackCheck", "Reproductor listo y reproduciendo. Marcando playback estable.")
                        onPlaybackStable(currentActiveUrl)
                    }
                }
            }
        }
    }

        val observer = LifecycleEventObserver { _, event ->
            when (event) {

                }

            }
        }
        lifecycleOwner.lifecycle.addObserver(observer)
        onDispose {
            lifecycleOwner.lifecycle.removeObserver(observer)

            override fun onTracksChanged(tracks: Tracks) {

                val videoGroupsCount = tracks.groups.count { it.type == C.TRACK_TYPE_VIDEO }

                if (videoGroupsCount == 0) {

                } else {

                                val bitrate = format.bitrate

                            }
                        }
                    }
                }

                Log.i("AudioDebug", "onTracksChanged detectó $audioGroupsCount grupos de audio y $videoGroupsCount grupos de video.")

                for ((groupIndex, group) in tracks.groups.withIndex()) {
                    if (group.type == C.TRACK_TYPE_AUDIO) {
                        for (i in 0 until group.length) {
                            val format = group.getTrackFormat(i)
                            val isSelected = group.isTrackSelected(i)
                            val mimeType = format.sampleMimeType ?: "unknown"
                            val codecs = format.codecs ?: "unknown"
                            val channelCount = format.channelCount

                            val bitrate = format.bitrate
                            val language = format.language ?: "und"

                            Log.i("AudioDebug", "Audio Group [$groupIndex] Track [$i]: sampleMimeType=$mimeType, codecs=$codecs, channelCount=$channelCount, sampleRate=$sampleRate, bitrate=$bitrate, language=$language, isSelected=$isSelected")
                        }
                    }
                }
            }

            override fun onAudioSessionIdChanged(audioSessionId: Int) {
                Log.i("AudioDebug", "onAudioSessionIdChanged: audioSessionId=$audioSessionId")
            }

            override fun onPlayerError(error: PlaybackException) {
                val errorCodeName = getExoPlayerErrorName(error.errorCode)
                Log.e("AudioDebug", "onPlayerError [Code: ${error.errorCode} - $errorCodeName], causeMessage=${error.cause?.message}: ${error.localizedMessage}", error)

                if (error.errorCode == PlaybackException.ERROR_CODE_BEHIND_LIVE_WINDOW) {
                    try {
                        exoPlayer.seekToDefaultPosition()
                        exoPlayer.prepare()
                        exoPlayer.play()
                        return
                    } catch (e: Exception) {
                        Log.e(TAG, "Error al intentar resincronizar BEHIND_LIVE_WINDOW", e)
                    }
                }

                val isNetworkOrDecoderError = error.errorCode in arrayOf(
                    PlaybackException.ERROR_CODE_IO_NETWORK_CONNECTION_FAILED,
                    PlaybackException.ERROR_CODE_IO_NETWORK_CONNECTION_TIMEOUT,
                    PlaybackException.ERROR_CODE_IO_BAD_HTTP_STATUS,
                    PlaybackException.ERROR_CODE_IO_FILE_NOT_FOUND,
                    PlaybackException.ERROR_CODE_DECODING_FAILED,
                    PlaybackException.ERROR_CODE_DECODER_INIT_FAILED
                ) || error.errorCode.toString().startsWith("20")

                if (isNetworkOrDecoderError && autoRetryCount < maxAutoRetries) {
                    autoRetryCount++
                    val backoffMillis = (autoRetryCount * 1500L).coerceAtMost(6000L)
                    coroutineScope.launch {
                        delay(backoffMillis)
                        try {
                            exoPlayer.prepare()
                            exoPlayer.playWhenReady = true
                        } catch (e: Exception) {
                            Log.e(TAG, "Error en reintento de reproducción #$autoRetryCount", e)
                        }
                    }
                } else {
                    switchToNextAlternative()
                }
            }
        })
    }

    LaunchedEffect(showControlsOverlay, isPlaying) {
        if (showControlsOverlay && isPlaying && !isPlaybackError && !isInactivePrompt) {
            delay(4000L)
            showControlsOverlay = false
        }
    }

    LaunchedEffect(Unit) {
        delay(100)
        playerFocusRequester.requestFocus()
    }

    LaunchedEffect(exoPlayer) {
        var lastPosition = -1L
        var freezeCounterSeconds = 0
        var hasStartedPlaying = false

        while (isActive) {
            delay(1000L)

            if (!isPlaybackError && !isInactivePrompt) {
                val playbackState = exoPlayer.playbackState
                val currentPos = exoPlayer.currentPosition
                val isPlayerPlaying = exoPlayer.isPlaying

                if (playbackState == Player.STATE_READY) {
                    hasStartedPlaying = true
                }

                val isPlayerBuffering = playbackState == Player.STATE_BUFFERING
                val isStuckReady = playbackState == Player.STATE_READY && (currentPos == lastPosition) && isPlayerPlaying

                if (isPlayerBuffering || isStuckReady) {
                    freezeCounterSeconds++
                    val threshold = if (!hasStartedPlaying) 15 else 6

                    if (freezeCounterSeconds >= threshold) {
                        freezeCounterSeconds = 0
                        autoRetryCount++

                        if (autoRetryCount > maxAutoRetries) {
                            switchToNextAlternative()
                        } else {
                            try {
                                if (isPlayerBuffering) {
                                    exoPlayer.prepare()
                                } else {
                                    exoPlayer.seekTo(currentPos)
                                }
                                exoPlayer.play()
                            } catch (e: Exception) {
                                Log.e(TAG, "Error durante la auto-recuperación de congelamiento", e)
                            }
                        }
                    }
                } else {
                    freezeCounterSeconds = 0
                    if (playbackState == Player.STATE_READY) {
                        autoRetryCount = 0
                    }
                }
                lastPosition = currentPos
            }
        }
    }

    LaunchedEffect(exoPlayer) {
        delay(2 * 60 * 60 * 1000L)
        if (exoPlayer.isPlaying) {
            exoPlayer.pause()
            isInactivePrompt = true
        }
    }

    LaunchedEffect(isPlaybackError) {
        if (isPlaybackError) {
            delay(100)
            retryFocusRequester.requestFocus()
        }
    }

    LaunchedEffect(isInactivePrompt) {
        if (isInactivePrompt) {
            delay(100)
            inactiveFocusRequester.requestFocus()
        }
    }

    Box(
        modifier = modifier
            .fillMaxSize()
            .focusRequester(playerFocusRequester)
            .focusable()
            .onKeyEvent { keyEvent ->
                if (keyEvent.type == KeyEventType.KeyDown) {
                    val keyCode = keyEvent.nativeKeyEvent.keyCode

                    when (keyCode) {
                        KeyEvent.KEYCODE_CHANNEL_UP, KeyEvent.KEYCODE_DPAD_UP -> {
                            if (channelsList.isNotEmpty() && currentChannelIndex != -1) {
                                val nextIndex = if (currentChannelIndex + 1 < channelsList.size) currentChannelIndex + 1 else 0
                                onChannelChange(channelsList[nextIndex])
                                showControlsOverlay = true
                                return@onKeyEvent true
                            }
                        }
                        KeyEvent.KEYCODE_CHANNEL_DOWN, KeyEvent.KEYCODE_DPAD_DOWN -> {
                            if (channelsList.isNotEmpty() && currentChannelIndex != -1) {
                                val prevIndex = if (currentChannelIndex - 1 >= 0) currentChannelIndex - 1 else channelsList.size - 1
                                onChannelChange(channelsList[prevIndex])
                                showControlsOverlay = true
                                return@onKeyEvent true
                            }
                        }
                        KeyEvent.KEYCODE_LAST_CHANNEL, KeyEvent.KEYCODE_MEDIA_PREVIOUS -> {
                            if (previousChannel != null) {
                                onChannelChange(previousChannel)
                                showControlsOverlay = true
                                return@onKeyEvent true
                            }
                        }
                        KeyEvent.KEYCODE_DPAD_CENTER,
                        KeyEvent.KEYCODE_ENTER,
                        KeyEvent.KEYCODE_NUMPAD_ENTER,
                        KeyEvent.KEYCODE_MEDIA_PLAY_PAUSE,
                        KeyEvent.KEYCODE_MEDIA_PLAY,
                        KeyEvent.KEYCODE_MEDIA_PAUSE -> {
                            if (exoPlayer.isPlaying) {
                                exoPlayer.pause()
                            } else {
                                exoPlayer.play()
                            }
                            showControlsOverlay = true
                            return@onKeyEvent true
                        }
                        KeyEvent.KEYCODE_DPAD_LEFT,
                        KeyEvent.KEYCODE_DPAD_RIGHT -> {
                            showControlsOverlay = true
                            return@onKeyEvent false
                        }
                    }
                }
                false
            }
    ) {
        AndroidView(
            modifier = Modifier.fillMaxSize(),
            factory = { ctx ->
                PlayerView(ctx).apply {
                    player = exoPlayer
                    useController = false
                    keepScreenOn = true
                }
            },
            update = { playerView ->
                playerView.player = exoPlayer
            }
        )

        if (isBuffering && !isPlaybackError && !isInactivePrompt) {
            Box(
                modifier = Modifier.fillMaxSize(),
                contentAlignment = Alignment.Center
            ) {
                TvCircularProgressIndicator(
                    color = MaterialTheme.colorScheme.primary,
                    strokeWidth = 4.dp,
                    modifier = Modifier.size(56.dp)
                )
            }
        }

        AnimatedVisibility(
            visible = showControlsOverlay && !isPlaybackError && !isInactivePrompt,
            enter = fadeIn(),
            exit = fadeOut(),
            modifier = Modifier.fillMaxSize()
        ) {
            Box(
                modifier = Modifier
                    .fillMaxSize()
                    .background(Color.Black.copy(alpha = 0.5f))
                    .padding(24.dp)
            ) {
                Row(
                    modifier = Modifier.align(Alignment.TopStart),
                    verticalAlignment = Alignment.CenterVertically,
                    horizontalArrangement = Arrangement.spacedBy(12.dp)
                ) {
                    Box(
                        modifier = Modifier
                            .background(Color.Red, shape = RoundedCornerShape(4.dp))
                            .padding(horizontal = 8.dp, vertical = 4.dp)
                    ) {
                        Text(
                            text = "EN VIVO",
                            color = Color.White,
                            fontSize = 12.sp,
                            fontWeight = FontWeight.Bold
                        )
                    }

                    Text(
                        text = channelName,
                        style = MaterialTheme.typography.titleLarge.copy(
                            fontWeight = FontWeight.Bold,
                            fontSize = 20.sp
                        ),
                        color = Color.White,
                        maxLines = 1,
                        overflow = TextOverflow.Ellipsis
                    )

                    if (channelsList.isNotEmpty() && currentChannelIndex != -1) {
                        Text(
                            text = "(${currentChannelIndex + 1}/${channelsList.size})",
                            style = MaterialTheme.typography.bodyMedium,
                            color = ElectricCyan,
                            fontWeight = FontWeight.Bold
                        )
                    }
                }

                Box(
                    modifier = Modifier.align(Alignment.Center)
                ) {
                    Surface(
                        modifier = Modifier.size(72.dp),
                        shape = RoundedCornerShape(36.dp)
                    ) {
                        Box(
                            contentAlignment = Alignment.Center,
                            modifier = Modifier.fillMaxSize()
                        ) {
                            Text(
                                text = if (isPlaying) "❚❚" else "▶",
                                color = Color.White,
                                fontSize = 28.sp,
                                fontWeight = FontWeight.Bold
                            )
                        }
                    }
                }

                Row(
                    modifier = Modifier
                        .align(Alignment.BottomCenter)
                        .fillMaxWidth()
                        .background(Color.Black.copy(alpha = 0.7f), shape = RoundedCornerShape(12.dp))
                        .padding(horizontal = 20.dp, vertical = 12.dp),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Row(horizontalArrangement = Arrangement.spacedBy(12.dp)) {
                        Button(
                            onClick = {
                                if (exoPlayer.isPlaying) {
                                    exoPlayer.pause()
                                } else {
                                    exoPlayer.play()
                                }
                                showControlsOverlay = true
                            }
                        ) {
                            Text(if (isPlaying) "Pausar" else "Reproducir")
                        }

                        if (previousChannel != null) {
                            Button(
                                onClick = {
                                    onChannelChange(previousChannel)
                                    showControlsOverlay = true
                                }
                            ) {
                                Text("↩ Volver a ${previousChannel.name}")
                            }
                        }
                    }

                    Row(
                        horizontalArrangement = Arrangement.spacedBy(12.dp),
                        verticalAlignment = Alignment.CenterVertically
                    ) {
                        Text(
                            text = "▲ / ▼ Cambiar canal",
                            style = MaterialTheme.typography.bodySmall,
                            color = Color.White.copy(alpha = 0.7f)
                        )

                        Button(
                            onClick = onBack
                        ) {
                            Text("Volver al Menú")
                        }
                    }
                }
            }
        }

        if (isPlaybackError) {
            Box(
                modifier = Modifier
                    .fillMaxSize()
                    .background(Color.Black.copy(alpha = 0.95f)),
                contentAlignment = Alignment.Center
            ) {
                Column(
                    horizontalAlignment = Alignment.CenterHorizontally,
                    modifier = Modifier
                        .widthIn(max = 550.dp)
                        .padding(24.dp)
                ) {
                    Text(
                        text = "Transmisión No Disponible",
                        style = MaterialTheme.typography.headlineMedium,
                        color = MaterialTheme.colorScheme.error,
                        textAlign = TextAlign.Center
                    )

                    Spacer(modifier = Modifier.height(12.dp))

                    Text(
                        text = errorMessageDetails,
                        style = MaterialTheme.typography.bodyMedium,
                        color = Color.White.copy(alpha = 0.85f),
                        textAlign = TextAlign.Center
                    )

                    Spacer(modifier = Modifier.height(28.dp))

                    Row(horizontalArrangement = Arrangement.spacedBy(16.dp)) {
                        Button(
                            onClick = {
                                autoRetryCount = 0
                                isPlaybackError = false
                                exoPlayer.prepare()
                                exoPlayer.playWhenReady = true
                            },
                            modifier = Modifier.focusRequester(retryFocusRequester)
                        ) {
                            Text("Reintentar")
                        }

                        Button(onClick = onBack) {
                            Text("Volver al Menú")
                        }
                    }
                }
            }
        }

        if (isInactivePrompt && !isPlaybackError) {
            Box(
                modifier = Modifier
                    .fillMaxSize()
                    .background(Color.Black.copy(alpha = 0.95f)),
                contentAlignment = Alignment.Center
            ) {
                Column(
                    horizontalAlignment = Alignment.CenterHorizontally,
                    modifier = Modifier
                        .widthIn(max = 550.dp)
                        .padding(24.dp)
                ) {
                    Text(
                        text = "¿Sigues viendo?",
                        style = MaterialTheme.typography.headlineMedium,
                        color = Color.White,
                        textAlign = TextAlign.Center
                    )

                    Spacer(modifier = Modifier.height(12.dp))

                    Text(
                        text = "Pausamos la transmisión automática para evitar el consumo innecesario de datos.",
                        style = MaterialTheme.typography.bodyLarge,
                        color = Color.White.copy(alpha = 0.8f),
                        textAlign = TextAlign.Center
                    )

                    Spacer(modifier = Modifier.height(28.dp))

                    Row(horizontalArrangement = Arrangement.spacedBy(16.dp)) {
                        Button(
                            onClick = {
                                isInactivePrompt = false
                                exoPlayer.play()
                            },
                            modifier = Modifier.focusRequester(inactiveFocusRequester)
                        ) {
                            Text("Continuar Viendo")
                        }

                        Button(onClick = onBack) {
                            Text("Salir al Menú")
                        }
                    }
                }
            }
        }
    }
}

@Composable
private fun AutoCloseableDisposable(player: ExoPlayer) {
    DisposableEffect(player) {
        onDispose {
            player.stop()
            player.release()
        }
    }
}

private fun getExoPlayerErrorName(errorCode: Int): String {
    return when (errorCode) {
        PlaybackException.ERROR_CODE_BEHIND_LIVE_WINDOW -> "BEHIND_LIVE_WINDOW"
        PlaybackException.ERROR_CODE_IO_NETWORK_CONNECTION_FAILED -> "IO_NETWORK_CONNECTION_FAILED"
        PlaybackException.ERROR_CODE_IO_NETWORK_CONNECTION_TIMEOUT -> "IO_NETWORK_CONNECTION_TIMEOUT"
        PlaybackException.ERROR_CODE_IO_INVALID_HTTP_CONTENT_TYPE -> "IO_INVALID_HTTP_CONTENT_TYPE"
        PlaybackException.ERROR_CODE_IO_BAD_HTTP_STATUS -> "IO_BAD_HTTP_STATUS"
        PlaybackException.ERROR_CODE_IO_FILE_NOT_FOUND -> "IO_FILE_NOT_FOUND"
        PlaybackException.ERROR_CODE_PARSING_CONTAINER_MALFORMED -> "PARSING_CONTAINER_MALFORMED"
        PlaybackException.ERROR_CODE_PARSING_MANIFEST_MALFORMED -> "PARSING_MANIFEST_MALFORMED"
        PlaybackException.ERROR_CODE_DECODING_FAILED -> "DECODING_FAILED"
        PlaybackException.ERROR_CODE_DECODER_INIT_FAILED -> "DECODER_INIT_FAILED"
        PlaybackException.ERROR_CODE_AUDIO_TRACK_INIT_FAILED -> "AUDIO_TRACK_INIT_FAILED"
        PlaybackException.ERROR_CODE_AUDIO_TRACK_WRITE_FAILED -> "AUDIO_TRACK_WRITE_FAILED"
        else -> "ERROR_CODE_$errorCode"
    }
}
